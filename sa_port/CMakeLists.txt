cmake_minimum_required(VERSION 3.15)
project(sa_port CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(sa_data src/ImgV2.cpp src/TextData.cpp src/BinaryIpl.cpp src/RwChunks.cpp src/StreamingWorld.cpp src/Streamer.cpp src/ColBounds.cpp src/Timecyc.cpp)
target_include_directories(sa_data PUBLIC src)
add_executable(streaming_ids_check src/streaming_ids_check.cpp)
add_executable(img_ls src/img_ls.cpp)
target_link_libraries(img_ls sa_data)
add_executable(load_all src/load_all.cpp)
target_link_libraries(load_all sa_data)
add_executable(stream_info src/stream_info.cpp)
target_link_libraries(stream_info sa_data)
add_executable(bnry_check src/bnry_check.cpp)
target_link_libraries(bnry_check sa_data)
add_executable(rw_check src/rw_check.cpp)
target_link_libraries(rw_check sa_data)

# librw (headless NULL platform) for real clump loading. Source: aap/librw, cloned under reVC/vendor.
set(SA_LIBRW_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/librw" CACHE PATH "librw source tree (clone https://github.com/aap/librw there)")
set(LIBRW_PLATFORM NULL CACHE STRING "" FORCE)
set(LIBRW_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBRW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${SA_LIBRW_DIR} librw_build)
add_executable(rw_load src/rw_load.cpp)
target_link_libraries(rw_load sa_data librw::librw)
